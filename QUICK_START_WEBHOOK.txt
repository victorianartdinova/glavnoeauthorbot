🎯 ЧТО НАДО СДЕЛАТЬ (5 минут):

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ШАГ 1: СОЗДАЙ SECRET НА VPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Запусти эту команду на VPS:

python3 << 'EOF'
import secrets
secret = secrets.token_urlsafe(32)
print(f"✅ SECRET: {secret}")
with open("/home/user/glavnoeauthorbot/.webhook_secret", "w") as f:
    f.write(secret)
print(f"✅ Сохранено в .webhook_secret")
EOF

👉 КОПИРУЙ ЗНАЧЕНИЕ SECRET - оно понадобится в GitHub!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ШАГ 2: УЗНАЙ IP СВОЕГО VPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

curl -s https://api.ipify.org
# или
hostname -I | awk '{print $1}'

👉 ЗАПОМНИ ЭТОТ IP!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ШАГ 3: ДОБАВЬ WEBHOOK В GITHUB
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Открой: https://github.com/victorianartdinova/glavnoeauthorbot
2. Settings → Webhooks → Add webhook
3. Заполни:
   Payload URL: http://YOUR_VPS_IP:8765/webhook
   Content type: application/json
   Secret: [ВСТАВЬ SECRET из ШАГа 1]
   Events: ✓ Just the push event
   Active: ✓ YES
4. Нажми "Add webhook"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ШАГ 4: ЗАПУСТИ WEBHOOK LISTENER НА VPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Запусти на VPS:

sudo cp /home/user/glavnoeauthorbot/scripts/glavnoe-webhook.service \
        /etc/systemd/system/

sudo systemctl daemon-reload
sudo systemctl start glavnoe-webhook
sudo systemctl enable glavnoe-webhook

Проверь что запустился:
sudo systemctl status glavnoe-webhook

Смотри логи:
sudo journalctl -u glavnoe-webhook -f

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ШАГ 5: ТЕСТИРУЙ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   tail -f /var/log/glavnoe-bot/auto_update.log
